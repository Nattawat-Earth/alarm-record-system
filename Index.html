    Google Sheets sync

    Paste the Web App URL you got after deploying the Apps Script (see the guide below). Without a URL, entries stay on this device only.

    How to connect Google Sheets (5 minutes)
    1. Open Google Sheets, create a sheet, name a tab Records.
    2. Add this header row in row 1:
      timestamp · date · shift · customer · model · mc · line · alarm · defect · issue · qty · downtime · remark · user
    3. Extensions → Apps Script. Replace the editor content with the snippet below, then click Deploy → New deployment → Web app. Execute as: Me. Who has access: Anyone. Copy the URL.
    4. Paste the URL into the field above and click Save, then Test connection.
    5. That's it. Every Save sends one row to Records. Use Resync pending to push any entries logged while offline.
    function doPost(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sh = ss.getSheetByName('Records') || ss.insertSheet('Records');
  if (sh.getLastRow() === 0) {
    sh.appendRow(['timestamp','date','shift','customer','model','mc','line',
                  'alarm','defect','issue','qty','downtime','remark','user']);
  }
  const d = JSON.parse(e.postData.contents);
  sh.appendRow([new Date(), d.date, d.shift, d.customer||'', d.model, d.mc||'',
                d.line, d.alarm, d.defect, d.issue||'',
                Number(d.qty)||0, Number(d.downtime)||0, d.remark||'', d.user||'']);
  return ContentService
    .createTextOutput(JSON.stringify({ok:true}))
    .setMimeType(ContentService.MimeType.JSON);
}
function doGet() {
  return ContentService
    .createTextOutput(JSON.stringify({ok:true, service:'breakdown-logger'}))
    .setMimeType(ContentService.MimeType.JSON);
}
